
using System;
using System.Windows;
using System.Windows.Threading;

namespace Learn_German_Basics
{
    public partial class App : Application
    {
        private SplashWindow? _splashWindow;
        private DispatcherTimer? _splashTimer;

        protected override void OnStartup(StartupEventArgs e)
        {
            base.OnStartup(e);

            // Show the splash screen first.
            _splashWindow = new SplashWindow();
            _splashWindow.Show();

            // Display the intro for approximately 3.2 seconds.
            _splashTimer = new DispatcherTimer
            {
                Interval = TimeSpan.FromMilliseconds(3200)
            };

            _splashTimer.Tick += SplashTimer_Tick;
            _splashTimer.Start();
        }

        private void SplashTimer_Tick(object? sender, EventArgs e)
        {
            _splashTimer?.Stop();
            _splashTimer = null;

            if (_splashWindow == null)
            {
                OpenHomeWindow();
                return;
            }

            _splashWindow.PlayOutro(OpenHomeWindow);
        }

        private void OpenHomeWindow()
        {
            // Prevent opening the home window more than once.
            if (MainWindow is not null &&
                MainWindow is not SplashWindow)
            {
                return;
            }

            var homeWindow = new MainWindow();

            // Register and display the real main window.
            MainWindow = homeWindow;
            ShutdownMode = ShutdownMode.OnMainWindowClose;

            homeWindow.Show();

            // Close the splash only after the home is visible.
            _splashWindow?.Close();
            _splashWindow = null;
        }
    }
}
